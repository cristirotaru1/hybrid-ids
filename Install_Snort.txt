Se instaleaza separat dnet:
  git clone https://github.com/dugsong/libdnet.git, si se instaleaza conform fisierului INSTALL

Se instaleaza libdaq:
  git clone https://github.com/snort3/libdaq.git

  cd libdaq
  ./bootstrap
  ./configure --prefix=/usr/local/lib/daq_s3
  make install

  cat /etc/ld.so.conf.d/libdaq3.conf
/usr/local/lib/daq_s3/lib/

  sudo ldconfig

Se instaleaza snort:
  git clone https://github.com/snort3/snort3.git
  cd snort3
  ./configure_cmake.sh --prefix=/path/to/snorty \
                       --with-daq-includes=/usr/local/lib/daq_s3/include/ \
                       --with-daq-libraries=/usr/local/lib/daq_s3/lib/

  cd build
  make -j $(nproc)
  make install
